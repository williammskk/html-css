public class TesteVisualStudioCode {
    public static void main(String[] args){

        // Escrever números de 0 a 10.

        int numeroMin = 0;
        for (int i = 0; i <= 10; i++){
            System.out.println(numeroMin);
            numeroMin += 1;
        }

        // Escrever números de 10 a 0.

        int numeroMax = 10;
        for (int i = 10; i >= 0 ; i--){
            System.out.println(numeroMax);
            numeroMax -= 1;
        }
    
    }
}
